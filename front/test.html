Выделение нескольких кнопок

Нажмите на кнопки, чтобы выделить их. Вы можете выбрать несколько элементов одновременно. Повторное нажатие снимает выделение.

Как это работает:

  • Нажмите на любую кнопку, чтобы выделить её
  • Нажмите на уже выделенную кнопку, чтобы снять выделение
  • Можно выделить любое количество кнопок одновременно
  • Используйте кнопку "Сбросить выделение", чтобы очистить все выбранные элементы

JavaScript код для выделения кнопок:

const buttons = document.querySelectorAll('.selectable-btn');
const selectedItemsContainer = document.getElementById('selectedItems');
const selectedCountElement = document.getElementById('selectedCount');
const clearButton = document.getElementById('clearAll');

let selectedItems = [];

// Обработчик клика по кнопке
buttons.forEach(button => {
    button.addEventListener('click', () => {
        button.classList.toggle('active');
        
        const value = button.getAttribute('data-value');
        
        if (button.classList.contains('active')) {
            // Добавляем кнопку в выбранные
            if (!selectedItems.includes(value)) {
                selectedItems.push(value);
            }
        } else {
            // Удаляем кнопку из выбранных
            selectedItems = selectedItems.filter(item => item !== value);
        }
        
        updateSelectionDisplay();
    });
});

// Функция обновления отображения выбранных элементов
function updateSelectionDisplay() {
    // Обновляем счетчик
    selectedCountElement.textContent = selectedItems.length;
    
    // Обновляем список выбранных элементов
    if (selectedItems.length === 0) {
        selectedItemsContainer.innerHTML = '<div class="empty-selection">Пока ничего не выбрано</div>';
    } else {
        selectedItemsContainer.innerHTML = selectedItems.map(item => 
            `<div class="selected-item">${item}</div>`
        ).join('');
    }
}

// Очистка всех выделений
clearButton.addEventListener('click', () => {
    buttons.forEach(button => {
        button.classList.remove('active');
    });
    selectedItems = [];
    updateSelectionDisplay();
});
